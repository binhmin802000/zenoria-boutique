import React, { useCallback, useEffect, useState } from "react";
import { AnimatePresence, motion } from "framer-motion";
import { X, Star, MessageCircle } from "lucide-react";
import { supabase } from "@/lib/supabase";
import { REVIEWS_RULES, REFUSAL_REASONS, reviewErrorMessage } from "@/lib/reviewsSettings";

// ============================================================
// PANNEAU « MES COMMANDES & AVIS » (page d'accueil) — ZENORIA_REVIEWS_STEP1_V1
//
// Même présentation que le panneau « Mes favoris ». Deux blocs :
//  - Bijoux à évaluer : fonction SQL list_reviewable_items() (commandes livrées de la cliente, délai de 30 jours) ;
//  - Mes avis : table reviews (la cliente ne lit que les siens), statut, motif de refus, réponse de la boutique.
// Dépôt : submit_review(), retrait : withdraw_review(). Tous les contrôles sont faits dans la base.
// Mode « étoiles seulement » : le champ commentaire disparaît (la base ignore aussi le texte).
// Interrupteur sur OFF : panneau réservé aux administratrices (tests), la base refuse les autres dépôts.
// ============================================================

const TIME_ZONE = "Europe/Paris";
const LABELS = ["", "Décevante", "Moyenne", "Bien", "Très bien", "Coup de cœur"];
const STATUS = {
  pending: { label: "En attente de relecture", cls: "bg-amber-50 text-amber-800" },
  published: { label: "Publié", cls: "bg-emerald-50 text-emerald-700" },
  refused: { label: "Non publié", cls: "bg-red-50 text-red-700" },
  withdrawn: { label: "Retiré", cls: "bg-[#f3f1f5] text-[#8b737e]" },
};

function formatDate(value) {
  if (!value) return "";
  return new Date(value).toLocaleDateString("fr-FR", { timeZone: TIME_ZONE });
}

function daysLeft(deadline) {
  const ms = new Date(deadline).getTime() - Date.now();
  return Math.max(0, Math.ceil(ms / 86400000));
}

function Stars({ value, size = 14 }) {
  return (
    <span className="inline-flex gap-0.5" aria-label={`${value} sur 5`}>
      {[1, 2, 3, 4, 5].map((i) => (
        <Star key={i} size={size} className={i <= value ? "fill-[#b48a5a] text-[#b48a5a]" : "text-[#e2d6da]"} />
      ))}
    </span>
  );
}

export default function MyReviewsPanel({ open, onClose, user, isAdmin, reviewsCfg }) {
  const enabled = Boolean(reviewsCfg && reviewsCfg.enabled);
  const starsOnly = Boolean(reviewsCfg && reviewsCfg.mode === "stars");
  const canUse = Boolean(user) && (enabled || isAdmin);

  const [items, setItems] = useState([]);
  const [reviews, setReviews] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [toast, setToast] = useState("");
  const [form, setForm] = useState(null); // { order_id, product_id, product_name }
  const [rating, setRating] = useState(0);
  const [comment, setComment] = useState("");
  const [sending, setSending] = useState(false);
  const [confirmId, setConfirmId] = useState(null);

  const load = useCallback(async () => {
    if (!canUse) return;
    setLoading(true);
    setError("");
    try {
      const [a, b] = await Promise.all([
        supabase.rpc("list_reviewable_items"),
        supabase
          .from("reviews")
          .select("id, order_id, product_id, rating, comment, status, refusal_reason, reply, reply_at, created_at, published_at, is_test, products(name)")
          .order("created_at", { ascending: false }),
      ]);
      if (a.error) throw a.error;
      if (b.error) throw b.error;
      setItems(a.data || []);
      setReviews(b.data || []);
    } catch (err) {
      setError(reviewErrorMessage(err));
    }
    setLoading(false);
  }, [canUse]);

  useEffect(() => {
    if (open) load();
    if (!open) {
      setForm(null);
      setConfirmId(null);
    }
  }, [open, load]);

  useEffect(() => {
    if (!toast) return undefined;
    const t = setTimeout(() => setToast(""), 4000);
    return () => clearTimeout(t);
  }, [toast]);

  const openForm = (it) => {
    setForm(it);
    setRating(0);
    setComment("");
    setError("");
  };

  const send = async () => {
    if (!form || rating < 1) return;
    setSending(true);
    setError("");
    try {
      const { error: err } = await supabase.rpc("submit_review", {
        p_order_id: form.order_id,
        p_product_id: form.product_id,
        p_rating: rating,
        p_comment: starsOnly ? null : comment,
      });
      if (err) throw err;
      setForm(null);
      setToast(`Merci ! Votre avis sera relu sous ${REVIEWS_RULES.moderationDays} jours ouvrés maximum.`);
      await load();
    } catch (err) {
      setError(reviewErrorMessage(err));
    }
    setSending(false);
  };

  const withdraw = async (id) => {
    setError("");
    try {
      const { error: err } = await supabase.rpc("withdraw_review", { p_review_id: id });
      if (err) throw err;
      setConfirmId(null);
      setToast("Votre avis a été retiré.");
      await load();
    } catch (err) {
      setError(reviewErrorMessage(err));
    }
  };

  const nameOf = (r) => {
    if (r.products && r.products.name) return r.products.name;
    const it = items.find((i) => i.order_id === r.order_id && i.product_id === r.product_id);
    return it ? it.product_name : "Bijou";
  };

  const maxChars = REVIEWS_RULES.maxCommentLength;

  return (
    <AnimatePresence>
      {open && (
        <>
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 bg-black/35 z-50" onClick={onClose} />
          <motion.aside
            initial={{ x: "100%" }}
            animate={{ x: 0 }}
            exit={{ x: "100%" }}
            style={{ height: "100dvh" }}
            data-testid="my-reviews-panel"
            className="fixed right-0 top-0 h-full w-full max-w-md bg-[#fbf8f4] z-50 shadow-2xl flex flex-col"
          >
            <div className="flex-none flex justify-between items-center px-5 py-4 border-b border-[#eadfe4]">
              <div className="min-w-0">
                <h3 className="font-serif text-2xl leading-none">Mes commandes &amp; avis</h3>
                <div className="text-xs text-[#8b737e] mt-1.5">Donnez votre avis sur les bijoux reçus</div>
              </div>
              <button onClick={onClose} className="h-10 w-10 rounded-full grid place-items-center hover:bg-[#f0e6ea]" aria-label="Fermer mes commandes et avis">
                <X />
              </button>
            </div>

            <div className="flex-1 min-h-0 overflow-y-auto px-5 py-4 space-y-5">
              {!canUse ? (
                <p className="text-sm text-[#70656a]">Les avis clientes ouvriront bientôt.</p>
              ) : (
                <>
                  {!enabled && isAdmin && (
                    <div className="rounded-xl bg-amber-50 text-amber-900 px-4 py-3 text-xs leading-5">
                      Interrupteur « Avis clientes » sur OFF : ce panneau n&apos;est visible que par l&apos;administration, pour les tests.
                    </div>
                  )}
                  {toast && <div data-testid="my-reviews-toast" className="rounded-xl bg-emerald-50 text-emerald-700 px-4 py-3 text-sm">{toast}</div>}
                  {error && <div role="alert" className="rounded-xl bg-red-50 text-red-600 px-4 py-3 text-sm">{error}</div>}

                  {/* BIJOUX À ÉVALUER */}
                  <section>
                    <h4 className="font-serif text-lg text-[#412f3b] mb-2">Bijoux à évaluer</h4>
                    {loading && items.length === 0 ? (
                      <p className="text-sm text-[#8b737e]">Chargement…</p>
                    ) : items.length === 0 ? (
                      <p className="text-sm text-[#70656a]">
                        Aucun bijou livré pour l&apos;instant. Vous pourrez donner votre avis pendant {REVIEWS_RULES.windowDays} jours après la livraison.
                      </p>
                    ) : (
                      <div className="space-y-2">
                        {items.map((it) => {
                          const st = it.review_status ? STATUS[it.review_status] : null;
                          const isOpen = form && form.order_id === it.order_id && form.product_id === it.product_id;
                          return (
                            <div key={`${it.order_id}-${it.product_id}`} className="bg-white rounded-2xl border border-[#eee2e5] p-3">
                              <div className="flex justify-between gap-2 items-start">
                                <div className="min-w-0">
                                  <div className="font-serif text-base leading-tight truncate">{it.product_name}</div>
                                  <div className="text-[11px] text-[#8b737e] mt-1">
                                    Commande n° {it.order_id} · livrée le {formatDate(it.delivered_at)}
                                    {it.can_review && ` · encore ${daysLeft(it.deadline)} jour${daysLeft(it.deadline) > 1 ? "s" : ""}`}
                                  </div>
                                </div>
                                {it.can_review && !isOpen ? (
                                  <button
                                    type="button"
                                    onClick={() => openForm(it)}
                                    data-testid={`review-open-${it.order_id}-${it.product_id}`}
                                    className="shrink-0 rounded-full bg-[#8f6075] hover:bg-[#71485b] text-white px-3 py-1 text-xs"
                                  >
                                    Donner mon avis
                                  </button>
                                ) : !it.can_review && st && (it.review_status === "pending" || it.review_status === "published") ? (
                                  <span className={`shrink-0 text-[10px] rounded-full px-2 py-0.5 ${st.cls}`}>{st.label}</span>
                                ) : !it.can_review ? (
                                  <span className="shrink-0 text-[10px] text-[#8b737e]">Délai dépassé</span>
                                ) : null}
                              </div>

                              {isOpen && (
                                <div className="mt-3 pt-3 border-t border-[#eee2e5]">
                                  <div className="flex gap-1" role="radiogroup" aria-label="Votre note">
                                    {[1, 2, 3, 4, 5].map((i) => (
                                      <button
                                        key={i}
                                        type="button"
                                        role="radio"
                                        aria-checked={rating === i}
                                        aria-label={`${i} étoile${i > 1 ? "s" : ""}`}
                                        onClick={() => setRating(i)}
                                        className="p-0.5"
                                      >
                                        <Star size={26} className={i <= rating ? "fill-[#b48a5a] text-[#b48a5a]" : "text-[#e2d6da]"} />
                                      </button>
                                    ))}
                                  </div>
                                  <div className="text-[11px] text-[#8b737e] mt-1">{rating ? `${rating}/5 · ${LABELS[rating]}` : "Choisissez une note (obligatoire)"}</div>

                                  {!starsOnly && (
                                    <div className="mt-3">
                                      <label htmlFor="review-comment" className="text-xs text-[#54434c]">Votre commentaire (facultatif)</label>
                                      <textarea
                                        id="review-comment"
                                        rows={4}
                                        maxLength={maxChars}
                                        value={comment}
                                        onChange={(e) => setComment(e.target.value)}
                                        placeholder="Qu'avez-vous pensé de votre bague ? Évitez les données personnelles (téléphone, adresse…)."
                                        className="mt-1 w-full rounded-xl border border-[#eadfe4] focus:border-[#8f6075] px-3 py-2 text-sm leading-6 outline-none bg-white"
                                      />
                                      <div className="text-right text-[11px] text-[#8b737e]">
                                        {comment.length.toLocaleString("fr-FR")} / {maxChars.toLocaleString("fr-FR")}
                                      </div>
                                    </div>
                                  )}

                                  <p className="text-[11px] text-[#8b737e] leading-4 mt-2">
                                    Votre avis sera publié avec votre prénom et l&apos;initiale de votre nom, après relecture sous {REVIEWS_RULES.moderationDays} jours ouvrés maximum. Il reste en ligne {REVIEWS_RULES.publicationYears} ans ; vous pouvez le retirer à tout moment.{" "}
                                    {enabled && (
                                      <a href="/charte-des-avis" target="_blank" rel="noopener noreferrer" className="underline text-[#8f6075]">Charte des avis</a>
                                    )}
                                  </p>
                                  <div className="flex gap-2 mt-3">
                                    <button
                                      type="button"
                                      onClick={send}
                                      disabled={sending || rating < 1}
                                      data-testid="review-send"
                                      className="rounded-full bg-[#8f6075] hover:bg-[#71485b] text-white px-4 h-9 text-sm disabled:opacity-50"
                                    >
                                      {sending ? "Envoi…" : "Envoyer mon avis"}
                                    </button>
                                    <button type="button" onClick={() => setForm(null)} className="rounded-full border border-[#d9c5cc] px-4 h-9 text-sm hover:bg-[#f5edf1]">
                                      Annuler
                                    </button>
                                  </div>
                                </div>
                              )}
                            </div>
                          );
                        })}
                      </div>
                    )}
                  </section>

                  {/* MES AVIS */}
                  <section>
                    <h4 className="font-serif text-lg text-[#412f3b] mb-2">Mes avis</h4>
                    {reviews.length === 0 ? (
                      <p className="text-sm text-[#70656a]">Vous n&apos;avez pas encore déposé d&apos;avis.</p>
                    ) : (
                      <div className="space-y-2">
                        {reviews.map((r) => {
                          const st = STATUS[r.status] || STATUS.pending;
                          return (
                            <div key={r.id} data-testid={`my-review-${r.id}`} className="bg-white rounded-2xl border border-[#eee2e5] p-3">
                              <div className="flex justify-between gap-2 items-start">
                                <div className="min-w-0">
                                  <div className="font-serif text-base leading-tight truncate">{nameOf(r)}</div>
                                  <div className="mt-1"><Stars value={r.rating} /></div>
                                </div>
                                <div className="shrink-0 flex flex-col items-end gap-1">
                                  <span className={`text-[10px] rounded-full px-2 py-0.5 ${st.cls}`}>{st.label}</span>
                                  {r.is_test && <span className="text-[10px] rounded-full px-2 py-0.5 bg-[#eef0f6] text-[#4a5578]">Test · jamais publié</span>}
                                </div>
                              </div>
                              {r.comment && <p className="text-sm text-[#54434c] leading-6 mt-2 whitespace-pre-line break-words">{r.comment}</p>}
                              <div className="text-[11px] text-[#8b737e] mt-1">
                                {r.status === "published" && r.published_at ? `Publié le ${formatDate(r.published_at)}` : `Envoyé le ${formatDate(r.created_at)}`}
                              </div>
                              {r.status === "refused" && (
                                <div className="text-[11px] text-red-700 mt-1 leading-4">
                                  Motif : {REFUSAL_REASONS[r.refusal_reason] || "non précisé"}. Vous pouvez déposer un nouvel avis tant que le délai de {REVIEWS_RULES.windowDays} jours court.
                                </div>
                              )}
                              {r.reply && r.status === "published" && (
                                <div className="mt-2 rounded-xl bg-[#f8f2f4] px-3 py-2">
                                  <div className="text-[11px] text-[#8f6075] flex items-center gap-1"><MessageCircle size={12} /> Réponse de Zenoria</div>
                                  <p className="text-sm text-[#54434c] leading-6 whitespace-pre-line break-words">{r.reply}</p>
                                </div>
                              )}
                              {(r.status === "pending" || r.status === "published") && (
                                <div className="mt-2">
                                  {confirmId === r.id ? (
                                    <div className="flex flex-wrap items-center gap-2 text-[11px]">
                                      <span className="text-[#54434c]">Retirer définitivement cet avis ?</span>
                                      <button type="button" onClick={() => withdraw(r.id)} className="rounded-full border border-red-200 text-red-700 px-3 py-1 hover:bg-red-50">
                                        Oui, retirer
                                      </button>
                                      <button type="button" onClick={() => setConfirmId(null)} className="underline text-[#8b737e]">
                                        Annuler
                                      </button>
                                    </div>
                                  ) : (
                                    <button type="button" onClick={() => setConfirmId(r.id)} className="text-[11px] underline text-[#8f6075]">
                                      Retirer mon avis
                                    </button>
                                  )}
                                </div>
                              )}
                            </div>
                          );
                        })}
                      </div>
                    )}
                  </section>
                </>
              )}
            </div>
          </motion.aside>
        </>
      )}
    </AnimatePresence>
  );
}
